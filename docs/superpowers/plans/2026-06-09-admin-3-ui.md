# Console admin — plan 3 : UI `/admin` (shell onglets + onglet config IA)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development. Steps en checkbox (`- [ ]`).

**Goal:** L'UI de la console admin : route `/admin` à onglets (v1 = onglet « IA »), formulaire de config IA (provider/baseURL/model + clé write-only + « Tester » + Enregistrer), gardée par `isAdmin`, et une entrée vers `/admin` visible seulement pour les admins. Termine la console.

**Architecture:** Éditeur (`editor/`):
- `lib/api/auth.ts` — ajouter `isAdmin` à `AuthUserDto` (vient déjà de `/me`).
- `lib/api/adminConfig.ts` — client (GET/PUT/test) en **cookie** (`credentials:"include"`).
- `features/admin/AssistantConfigForm.tsx` — le formulaire.
- `routes/admin.tsx` — page `/admin` (shell onglets, gardée `isAdmin`).
- `routes/account.tsx` — lien vers `/admin` si `isAdmin`.
- `locales/{en,fr}.ts` — namespace `admin.*`.

**Tech Stack:** React 19, TanStack Router (file-based → `npm run generate-routes`), Vitest + testing-library, `sonner`, radix Tabs (dep). Auth admin = **cookie `artisan_user_session`** (les endpoints `/api/v1/admin/*` sont gardés `userAuth`+`requireAdmin`).

**Prérequis :** plans 1+2 mergés (`/me` renvoie `isAdmin` ; endpoints `/api/v1/admin/assistant-config`).

```ts
const apiUrl = (): string => (import.meta.env.VITE_API_URL as string | undefined) ?? "http://localhost:3001";

export interface AssistantConfigDto { provider: string; baseUrl: string; model: string; keySet: boolean }

export async function getAssistantConfig(): Promise<AssistantConfigDto | null> {
  const res = await fetch(`${apiUrl()}/api/v1/admin/assistant-config`, { credentials: "include" });
  if (!res.ok) return null;
  return (await res.json()) as AssistantConfigDto;
}

export async function putAssistantConfig(body: {
  provider: string; baseUrl: string; model: string; apiKey?: string;
}): Promise<{ ok: true; data: AssistantConfigDto } | { ok: false; status: number }> {
  const res = await fetch(`${apiUrl()}/api/v1/admin/assistant-config`, {
    method: "PUT", credentials: "include",
    headers: { "Content-Type": "application/json" }, body: JSON.stringify(body),
  });
  if (!res.ok) return { ok: false, status: res.status };
  return { ok: true, data: (await res.json()) as AssistantConfigDto };
}

export async function testAssistantConfig(body: {
  provider: string; baseUrl: string; model: string; apiKey?: string;
}): Promise<{ ok: boolean; error?: string }> {
  const res = await fetch(`${apiUrl()}/api/v1/admin/assistant-config/test`, {
    method: "POST", credentials: "include",
    headers: { "Content-Type": "application/json" }, body: JSON.stringify(body),
  });
  if (!res.ok) return { ok: false, error: `http_${res.status}` };
  return (await res.json()) as { ok: boolean; error?: string };
}
```

- [ ] **Step 1 : `AssistantConfigForm.tsx`** :
```tsx
import { useEffect, useState } from "react";
import { useTranslation } from "react-i18next";
import { toast } from "sonner";
import { getAssistantConfig, putAssistantConfig, testAssistantConfig } from "@/lib/api/adminConfig";

const PROVIDERS = ["openai", "deepseek", "mistral", "openrouter", "local"];

export function AssistantConfigForm() {
  const { t } = useTranslation();
  const [provider, setProvider] = useState("openai");
  const [baseUrl, setBaseUrl] = useState("");
  const [model, setModel] = useState("");
  const [apiKey, setApiKey] = useState("");
  const [keySet, setKeySet] = useState(false);
  const [busy, setBusy] = useState(false);

  useEffect(() => {
    getAssistantConfig().then((c) => {
      if (!c) return;
      setProvider(c.provider); setBaseUrl(c.baseUrl); setModel(c.model); setKeySet(c.keySet);
    });
  }, []);

  const body = () => ({ provider, baseUrl, model, ...(apiKey ? { apiKey } : {}) });

  async function save() {
    setBusy(true);
    const r = await putAssistantConfig(body());
    setBusy(false);
    if (r.ok) { setKeySet(r.data.keySet); setApiKey(""); toast.success(t("admin.ai.saved")); }
    else toast.error(t("admin.ai.saveFail"));
  }
  async function test() {
    setBusy(true);
    const r = await testAssistantConfig(body());
    setBusy(false);
    if (r.ok) toast.success(t("admin.ai.testOk"));
    else toast.error(t("admin.ai.testFail", { error: r.error ?? "" }));
  }

  return (
    <form className="flex max-w-lg flex-col gap-4" onSubmit={(e) => { e.preventDefault(); void save(); }}>
      <h2 className="eyebrow">{t("admin.ai.heading")}</h2>
      <label>{t("admin.ai.provider")}
        <select value={provider} onChange={(e) => setProvider(e.target.value)} aria-label={t("admin.ai.provider")}>
          {PROVIDERS.map((p) => <option key={p} value={p}>{p}</option>)}
        </select>
      </label>
      <label>{t("admin.ai.baseUrl")}
        <input value={baseUrl} onChange={(e) => setBaseUrl(e.target.value)} aria-label={t("admin.ai.baseUrl")} />
      </label>
      <label>{t("admin.ai.model")}
        <input value={model} onChange={(e) => setModel(e.target.value)} aria-label={t("admin.ai.model")} />
      </label>
      <label>{t("admin.ai.apiKey")}
        <input type="password" value={apiKey} onChange={(e) => setApiKey(e.target.value)}
          placeholder={keySet ? t("admin.ai.keySetPlaceholder") : ""} aria-label={t("admin.ai.apiKey")} />
      </label>
      <div className="flex gap-2">
        <button type="button" disabled={busy} onClick={() => void test()}>{t("admin.ai.test")}</button>
        <button type="submit" disabled={busy}>{t("admin.ai.save")}</button>
      </div>
    </form>
  );
}
```

```tsx
import { createFileRoute } from "@tanstack/react-router";
import { useTranslation } from "react-i18next";
import { useAuthStore } from "@/state/authStore";
import { AssistantConfigForm } from "@/features/admin/AssistantConfigForm";

export const Route = createFileRoute("/admin")({ component: AdminPage });

function AdminPage() {
  const { t } = useTranslation();
  const user = useAuthStore((s) => s.currentUser);
  const loading = useAuthStore((s) => s.isLoading);
  if (loading) return null;
  if (!user?.isAdmin) {
    return <div className="flex h-full items-center justify-center p-8 text-[var(--text-mut)]">{t("admin.forbidden")}</div>;
  }
  return (
    <div className="mx-auto flex max-w-3xl flex-col gap-6 px-8 py-12">
      <h1 className="display text-[32px]" style={{ fontFamily: "var(--font-display)" }}>{t("admin.title")}</h1>
      {/* v1: a single "IA" tab. Add more tabs (users, logs) here later. */}
      <nav className="flex gap-2 border-b border-[var(--border-soft)]"><span className="border-b-2 border-[var(--brand)] pb-2 text-sm">{t("admin.tabs.ai")}</span></nav>
      <AssistantConfigForm />
    </div>
  );
}
```
