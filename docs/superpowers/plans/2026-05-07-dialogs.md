# Dialog System Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add Paper Dialog support so pack authors can define `dialog/v1` YAML files with optional form inputs; dialogs open via `open_dialog` action and are declared in the pack manifest.

**Architecture:** Single unified `dialog/v1` schema mirrors the menu/command pattern. `BundleDialog` wraps YAML + langMaps. `DialogRenderer` builds Paper `Dialog` objects with one-shot inline callbacks per button; when a button fires, input values from `DialogResponseView` are injected into `ctx.args` before executing the button's `ClickHandler`. `SyncedMenuProvider` gains a `dialogs` map and an `executorProvider: () -> ActionExecutor` lazy-lambda (same pattern as `onReload`) to break the circular dependency between provider and executor.

**Tech Stack:** Kotlin 2.0, Paper API 1.21.7+ (Dialog API), existing ActionExecutor / ExecutionContext / BundleCompiler / ConditionParser / ActionParser infrastructure.

Create `plugin/src/main/kotlin/net/artisanmc/plugin/render/DialogRenderer.kt`:

```kotlin
package net.artisanmc.plugin.render

import net.artisanmc.plugin.engine.*
import net.artisanmc.plugin.model.BundleDialog
import net.artisanmc.plugin.model.DialogInputYaml
import io.papermc.paper.dialog.DialogResponseView
import io.papermc.paper.registry.data.dialog.ActionButton
import io.papermc.paper.registry.data.dialog.Dialog
import io.papermc.paper.registry.data.dialog.DialogBase
import io.papermc.paper.registry.data.dialog.action.DialogAction
import io.papermc.paper.registry.data.dialog.body.DialogBody
import io.papermc.paper.registry.data.dialog.input.DialogInput
import io.papermc.paper.registry.data.dialog.type.DialogType
import net.kyori.adventure.audience.Audience
import net.kyori.adventure.chat.ClickCallback
import net.kyori.adventure.text.minimessage.MiniMessage
import org.bukkit.Material
import org.bukkit.entity.Player
import org.bukkit.inventory.ItemStack

object DialogRenderer {

    private val mm = MiniMessage.miniMessage()

    fun open(player: Player, dialog: BundleDialog, lang: String, ctx: ExecutionContext, executor: ActionExecutor) {
        val langMap = dialog.langMaps[lang] ?: emptyMap()
        val ctxWithLang = ctx.copy(lang = lang, langMap = langMap)

        val bodyComponents = dialog.yaml.body.mapNotNull { b ->
            when (b.type) {
                "text" -> b.text?.let { DialogBody.plainMessage(mm.deserialize(resolveI18n(it, langMap))) }
                "item" -> b.material?.let { mat ->
                    runCatching { Material.valueOf(mat.uppercase()) }.getOrNull()?.let { material ->
                        DialogBody.item(ItemStack(material, b.count.coerceAtLeast(1)))
                    }
                }
                else -> null
            }
        }

        val inputList = dialog.yaml.inputs.mapNotNull { inputYaml ->
            val label = mm.deserialize(resolveI18n(inputYaml.label, langMap))
            runCatching {
                when (inputYaml.type) {
                    "text" -> DialogInput.text(inputYaml.id) { b ->
                        b.label(label)
                        b.maxLength(inputYaml.maxLength)
                        b.defaultValue(inputYaml.default)
                    }
                    "bool" -> DialogInput.bool(inputYaml.id) { b ->
                        b.label(label)
                        b.defaultValue(inputYaml.default.toBooleanStrictOrNull() ?: false)
                    }
                    "number_range" -> DialogInput.numberRange(inputYaml.id) { b ->
                        b.label(label)
                        b.min(inputYaml.min.toFloat())
                        b.max(inputYaml.max.toFloat())
                        b.defaultValue((inputYaml.default.toDoubleOrNull() ?: inputYaml.min).toFloat())
                    }
                    "single_option" -> DialogInput.singleOption(inputYaml.id) { b ->
                        b.label(label)
                        inputYaml.options.forEach { opt ->
                            b.addOption(opt.value, mm.deserialize(resolveI18n(opt.label, langMap)))
                        }
                        b.defaultValue(inputYaml.default)
                    }
                    else -> null
                }
            }.getOrNull()
        }

        val inputIds = dialog.yaml.inputs.map { it.id }
        val inputsByIdYaml = dialog.yaml.inputs.associateBy { it.id }

        val buttons = dialog.yaml.buttons.map { btnYaml ->
            val requirements = btnYaml.requirements.map { ConditionParser.parse(it) }
            val actions      = btnYaml.actions.map { ActionParser.parse(it) }
            val fallback     = btnYaml.fallback.map { ActionParser.parse(it) }
            val handler      = ClickHandler(requirements, actions, fallback)
            val btnLabel     = mm.deserialize(resolveI18n(btnYaml.label, langMap))

            ActionButton.builder(
                btnLabel,
                DialogAction.customClick(
                    { view: DialogResponseView, _: Audience ->
                        val inputValues: Map<String, Any> = inputIds.associateWith { id ->
                            extractInputValue(view, id, inputsByIdYaml[id])
                        }
                        val newCtx = ctxWithLang.copy(args = ctxWithLang.args + inputValues)
                        executor.handle(handler, newCtx)
                    },
                    ClickCallback.Options.builder()
                        .uses(1)
                        .lifetime(ClickCallback.DEFAULT_LIFETIME)
                        .build()
                )
            ).build()
        }

        val title = mm.deserialize(resolveI18n(dialog.yaml.title, langMap))
        val base = DialogBase.builder(title)
            .canCloseWithEscape(dialog.yaml.canCloseWithEscape)
            .body(bodyComponents)
            .inputs(inputList)
            .build()

        val type = if (buttons.size == 1) DialogType.notice(buttons[0]) else DialogType.multiAction(buttons)

        player.showDialog(Dialog.create { b -> b.base(base).type(type) })
    }

    private fun extractInputValue(view: DialogResponseView, id: String, inputYaml: DialogInputYaml?): String =
        when (inputYaml?.type) {
            "text"          -> view.getString(id) ?: inputYaml.default
            "bool"          -> (view.getBoolean(id)?.toString()) ?: inputYaml.default
            "number_range"  -> {
                val f = view.getFloat(id)?.toDouble() ?: (inputYaml.default.toDoubleOrNull() ?: inputYaml.min)
                if (f == f.toLong().toDouble()) f.toLong().toString() else f.toString()
            }
            "single_option" -> view.getString(id) ?: inputYaml.default
            else            -> ""
        }

    private fun resolveI18n(raw: String, langMap: Map<String, String>): String {
        if (!raw.startsWith("{i18n:")) return raw
        val key = raw.removePrefix("{i18n:").removeSuffix("}")
        return langMap[key] ?: raw
    }
}
```

private fun reloadFromDisk() {
