# Plugin i18n System Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add an i18n system to the plugin with YAML lang files (MiniMessage format), a `language` config option, English fallback, and optional per-player language detection via `Player.locale()`.

**Architecture:** `LangManager` holds a `Map<lang, Map<key, MiniMessageString>>` loaded from JAR resources (`lang/en.yml`, `lang/fr.yml`) with per-plugin-folder overrides. It exposes `get(key, lang)→Component` and `getRaw(key, lang)→String` with a fallback chain: requested lang → `defaultLang` → `"en"` → key itself. All hardcoded strings in `ArgParser`, `PackCommand`, `SyncedMenuProvider`, and `ArtisanCommand` are replaced with `langManager.get(...)` calls. `ArtisanPlugin` creates `LangManager` from `PluginConfig.language` / `PluginConfig.perPlayerLanguage` and injects it into every consumer.

**Tech Stack:** Kotlin 2.0, Paper 1.21.7, MockBukkit; Bukkit `YamlConfiguration` for loading lang files; Adventure MiniMessage for text; `kotlin.test`.

```kotlin
package net.artisanmc.plugin.lang

import net.kyori.adventure.text.Component
import net.kyori.adventure.text.minimessage.MiniMessage
import org.bukkit.command.CommandSender
import org.bukkit.configuration.file.YamlConfiguration
import org.bukkit.entity.Player
import org.bukkit.plugin.Plugin
import java.io.File
import java.io.InputStreamReader

class LangManager(
    private val translations: Map<String, Map<String, String>>,
    private val defaultLang: String,
    private val perPlayerLanguage: Boolean,
) {
    private val mm = MiniMessage.miniMessage()

    companion object {
        fun fromPlugin(plugin: Plugin, defaultLang: String, perPlayerLanguage: Boolean): LangManager {
            val translations = mutableMapOf<String, Map<String, String>>()
            for (lang in listOf("en", "fr")) {
                val map = loadLang(plugin, lang)
                if (map != null) translations[lang] = map
            }
            return LangManager(translations, defaultLang, perPlayerLanguage)
        }

        private fun loadLang(plugin: Plugin, lang: String): Map<String, String>? {
            val override = File(plugin.dataFolder, "lang/$lang.yml")
            val cfg = if (override.exists()) {
                YamlConfiguration.loadConfiguration(override)
            } else {
                val stream = plugin.getResource("lang/$lang.yml") ?: return null
                YamlConfiguration.loadConfiguration(InputStreamReader(stream, Charsets.UTF_8))
            }
            return cfg.getKeys(true)
                .filter { cfg.isString(it) }
                .associateWith { cfg.getString(it)!! }
        }
    }

    fun getRaw(key: String, lang: String = defaultLang, vararg args: Pair<String, Any>): String {
        val effective = if (translations.containsKey(lang)) lang else defaultLang
        val value = translations[effective]?.get(key)
            ?: translations["en"]?.get(key)
            ?: key
        return args.fold(value) { s, (k, v) -> s.replace("{$k}", v.toString()) }
    }

    fun get(key: String, lang: String = defaultLang, vararg args: Pair<String, Any>): Component =
        mm.deserialize(getRaw(key, lang, *args))

    fun playerLang(sender: CommandSender): String {
        if (!perPlayerLanguage || sender !is Player) return defaultLang
        return try {
            val lang = sender.locale().language.lowercase()
            if (translations.containsKey(lang)) lang else defaultLang
        } catch (_: Exception) { defaultLang }
    }

    fun get(key: String, sender: CommandSender, vararg args: Pair<String, Any>): Component =
        get(key, playerLang(sender), *args)

    fun getRaw(key: String, sender: CommandSender, vararg args: Pair<String, Any>): String =
        getRaw(key, playerLang(sender), *args)
}
```

```kotlin
package net.artisanmc.plugin

import net.artisanmc.plugin.command.CommandRegistrar
import net.artisanmc.plugin.command.ArtisanCommand
import net.artisanmc.plugin.config.PluginConfig
import net.artisanmc.plugin.engine.*
import net.artisanmc.plugin.lang.LangManager
import net.artisanmc.plugin.provider.MenuProvider
import net.artisanmc.plugin.provider.SyncedMenuProvider
import net.artisanmc.plugin.render.MenuListener
import org.bukkit.plugin.java.JavaPlugin
import java.io.File

open class ArtisanPlugin : JavaPlugin() {
    lateinit var menuProvider: MenuProvider
        private set
    private lateinit var ifzCommand: ArtisanCommand
    private lateinit var executor: ActionExecutor
    private lateinit var sessions: MenuSessionStore
    private lateinit var cooldowns: CooldownStore
    private lateinit var commandRegistrar: CommandRegistrar
    lateinit var langManager: LangManager
        private set

    override fun onEnable() {
        saveDefaultConfig()
        val cfg = PluginConfig.from(config)
        val serverRoot = dataFolder.parentFile.parentFile
        val packsDir = File(serverRoot, cfg.packsDir).also { it.mkdirs() }

        langManager = LangManager.fromPlugin(this, cfg.language, cfg.perPlayerLanguage)

        val softDeps = SoftDepRegistry(this)
        cooldowns = CooldownStore()
        sessions = MenuSessionStore()
        val evaluator = ConditionEvaluator(softDeps, cooldowns)

        commandRegistrar = CommandRegistrar(this, langManager)
        menuProvider = SyncedMenuProvider(
            packsDir         = packsDir,
            serverKey        = cfg.serverKey,
            wsUrl            = cfg.wsUrl,
            logger           = logger,
            evaluator        = evaluator,
            sessions         = sessions,
            langManager      = langManager,
            onReload         = { commandRegistrar.reload(menuProvider.allBundles(), executor, evaluator) },
            executorProvider = { executor },
        )
        menuProvider.load()

        executor = ActionExecutor(this, menuProvider, sessions, softDeps, cooldowns, evaluator)
        commandRegistrar.reload(menuProvider.allBundles(), executor, evaluator)

        ifzCommand = ArtisanCommand(this, langManager)
        getCommand("ifz")?.setExecutor(ifzCommand)
        getCommand("ifz")?.tabCompleter = ifzCommand
        server.pluginManager.registerEvents(MenuListener(sessions, executor), this)
        logger.info("Artisan ${description.version} enabled")
    }

    override fun onDisable() {
        ifzCommand.shutdown()
        commandRegistrar.unregisterAll()
        menuProvider.unload()
        logger.info("Artisan disabled")
    }

    fun reinit(cfg: PluginConfig) {
        langManager = LangManager.fromPlugin(this, cfg.language, cfg.perPlayerLanguage)
        menuProvider.unload()
        val packsDir = File(dataFolder.parentFile.parentFile, cfg.packsDir).also { it.mkdirs() }

        val softDeps = SoftDepRegistry(this)
        val evaluator = ConditionEvaluator(softDeps, cooldowns)

        menuProvider = SyncedMenuProvider(
            packsDir         = packsDir,
            serverKey        = cfg.serverKey,
            wsUrl            = cfg.wsUrl,
            logger           = logger,
            evaluator        = evaluator,
            sessions         = sessions,
            langManager      = langManager,
            onReload         = { commandRegistrar.reload(menuProvider.allBundles(), executor, evaluator) },
            executorProvider = { executor },
        )
        menuProvider.load()
        executor = ActionExecutor(this, menuProvider, sessions, softDeps, cooldowns, evaluator)
        commandRegistrar.reload(menuProvider.allBundles(), executor, evaluator)
    }
}
```

```kotlin
package net.artisanmc.plugin.command

import net.artisanmc.plugin.lang.LangManager
import org.bukkit.Bukkit
import org.bukkit.Material
import org.bukkit.command.CommandSender
import org.bukkit.entity.Player

object ArgParser {

    fun parse(
        defs: List<ArgDef>,
        args: Array<String>,
        sender: CommandSender,
        lang: LangManager,
    ): Map<String, Any>? {
        val result = mutableMapOf<String, Any>()
        for ((index, def) in defs.withIndex()) {
            val rawValue: String? = when {
                index < args.size && index == defs.lastIndex ->
                    args.drop(index).joinToString(" ")
                index < args.size -> args[index]
                else -> null
            }

            if (rawValue == null) {
                if (def.required) {
                    sender.sendMessage(lang.get("arg.missing", sender, "arg" to def.name))
                    return null
                }
                val default = when (def) {
                    is ArgDef.StringArg -> def.default
                    is ArgDef.IntArg    -> def.default
                    else                -> null
                }
                if (default != null) result[def.name] = default
                continue
            }

            val parsed: Any = when (def) {
                is ArgDef.StringArg      -> rawValue
                is ArgDef.IntArg         -> rawValue.toIntOrNull() ?: run {
                    sender.sendMessage(lang.get("arg.not_integer", sender, "arg" to def.name, "value" to rawValue))
                    return null
                }
                is ArgDef.PlayerArg      -> Bukkit.getPlayerExact(rawValue) ?: run {
                    sender.sendMessage(lang.get("arg.player_offline", sender, "value" to rawValue))
                    return null
                }
                is ArgDef.WorldArg       -> Bukkit.getWorld(rawValue) ?: run {
                    sender.sendMessage(lang.get("arg.world_not_loaded", sender, "value" to rawValue))
                    return null
                }
                is ArgDef.ItemMaterialArg -> Material.matchMaterial(rawValue) ?: run {
                    sender.sendMessage(lang.get("arg.unknown_material", sender, "value" to rawValue))
                    return null
                }
                is ArgDef.EnumArg        -> {
                    val match = def.values.firstOrNull { it.equals(rawValue, ignoreCase = true) } ?: run {
                        sender.sendMessage(lang.get("arg.invalid_enum", sender,
                            "arg" to def.name, "value" to rawValue,
                            "values" to def.values.joinToString(", ")))
                        return null
                    }
                    match
                }
            }
            result[def.name] = parsed
        }
        return result
    }

    fun tabComplete(defs: List<ArgDef>, args: Array<String>): List<String> {
        val index = (args.size - 1).coerceAtLeast(0)
        val def = defs.getOrNull(index) ?: return emptyList()
        val prefix = args.lastOrNull() ?: ""
        return when (def) {
            is ArgDef.EnumArg   -> def.values.filter { it.startsWith(prefix, ignoreCase = true) }
            is ArgDef.PlayerArg -> Bukkit.getOnlinePlayers()
                .map { it.name }
                .filter { it.startsWith(prefix, ignoreCase = true) }
            is ArgDef.WorldArg  -> Bukkit.getWorlds()
                .map { it.name }
                .filter { it.startsWith(prefix, ignoreCase = true) }
            is ArgDef.ItemMaterialArg -> Material.entries
                .map { it.name.lowercase() }
                .filter { it.startsWith(prefix, ignoreCase = true) }
            else -> emptyList()
        }
    }
}
```

```kotlin
package net.artisanmc.plugin.command

import net.artisanmc.plugin.engine.ActionExecutor
import net.artisanmc.plugin.engine.ConditionEvaluator
import net.artisanmc.plugin.lang.LangManager
import net.artisanmc.plugin.model.PackBundle
import org.bukkit.Bukkit
import org.bukkit.command.SimpleCommandMap
import org.bukkit.plugin.java.JavaPlugin
import java.util.logging.Logger

class CommandRegistrar(
    private val plugin: JavaPlugin,
    private val lang: LangManager,
) {
    private val logger: Logger get() = plugin.logger
    private val registered = mutableListOf<String>()

    fun reload(bundles: List<PackBundle>, executor: ActionExecutor, evaluator: ConditionEvaluator) {
        unregisterAll()
        val commandMap = Bukkit.getCommandMap()
        val namespace = plugin.name.lowercase()
        val seen = mutableSetOf<String>()
        val totalCmds = bundles.sumOf { it.commands.size }
        logger.info("[IFZ-DEBUG] CommandRegistrar.reload: ${bundles.size} bundles, $totalCmds commands total")

        for (bundle in bundles) {
            logger.info("[IFZ-DEBUG] Bundle ${bundle.packId}: ${bundle.commands.size} commands: ${bundle.commands.map { it.name }}")
            for (cmdYaml in bundle.commands) {
                val name = cmdYaml.name.lowercase()
                logger.info("[IFZ-DEBUG] Processing command '$name' actions=${cmdYaml.resolvedActions().size} steps=${cmdYaml.steps.size}")

                if (name in seen) {
                    logger.warning("Command '$name' defined in multiple packs — only first registration kept.")
                    continue
                }

                val existing = commandMap.getCommand(name)
                if (existing != null && existing !is PackCommand) {
                    logger.warning("Command '$name' conflicts with an existing command — skipped.")
                    continue
                }

                val compiled = CommandCompiler.compile(cmdYaml, logger)
                if (compiled == null) {
                    logger.warning("[IFZ-DEBUG] CommandCompiler returned null for '$name'")
                    continue
                }
                compiled.aliases.forEach { alias ->
                    val existingAlias = commandMap.getCommand(alias.lowercase())
                    if (existingAlias != null && existingAlias !is PackCommand) {
                        logger.warning("Alias '/$alias' for command '/$name' conflicts with an existing command — alias may not work.")
                    }
                }
                val packCommand = PackCommand(compiled, executor, lang)
                commandMap.register(namespace, packCommand)
                registered += name
                seen += name
                compiled.aliases.forEach { alias ->
                    val normalised = alias.lowercase()
                    registered += normalised
                    seen += normalised
                }
                logger.info("[IFZ-DEBUG] Registered command: /$name (actions=${compiled.handler.actions.size})")
            }
        }
    }

    fun unregisterAll() {
        if (registered.isEmpty()) return
        val commandMap = Bukkit.getCommandMap()
        registered.forEach { name ->
            commandMap.getCommand(name)?.unregister(commandMap)
            commandMap.getCommand("${plugin.name.lowercase()}:$name")?.unregister(commandMap)
        }
        runCatching {
            val knownCommands = (commandMap as? SimpleCommandMap)
                ?.let { SimpleCommandMap::class.java.getDeclaredField("knownCommands").also { f -> f.isAccessible = true }.get(it) as? MutableMap<*, *> }
            @Suppress("UNCHECKED_CAST")
            (knownCommands as? MutableMap<String, Any>)?.keys?.removeAll { key ->
                registered.any { name -> key == name || key == "${plugin.name.lowercase()}:$name" }
            }
        }
        registered.clear()
    }
}
```

```kotlin
import net.kyori.adventure.text.format.NamedTextColor
import net.kyori.adventure.text.format.TextDecoration
import org.bukkit.Bukkit
import org.bukkit.command.Command
import org.bukkit.command.CommandExecutor
import org.bukkit.command.CommandSender
import org.bukkit.command.ConsoleCommandSender
import org.bukkit.command.TabCompleter
import org.bukkit.entity.Player
import net.kyori.adventure.text.Component

class ArtisanCommand(
    private val plugin: ArtisanPlugin,
    private val lang: LangManager,
) : CommandExecutor, TabCompleter {

    @Volatile private var setupClient: SetupClient? = null

    fun shutdown() {
        setupClient?.disconnect()
        setupClient = null
    }

    override fun onTabComplete(
        sender: CommandSender, command: Command, label: String, args: Array<out String>
    ): List<String> {
        return when {
            args.size == 1 -> listOf("menu", "reload", "sync", "setup", "editor", "trusteditor")
                .filter { it.startsWith(args[0].lowercase()) }
            args.size == 2 && args[0].equals("menu", ignoreCase = true) ->
                plugin.menuProvider.listMenuIds().filter { it.startsWith(args[1]) }
            args.size == 2 && args[0].equals("sync", ignoreCase = true) ->
                listOf("status", "reject").filter { it.startsWith(args[1].lowercase()) }
            args.size == 2 && args[0].equals("editor", ignoreCase = true) ->
                listOf("revoke").filter { it.startsWith(args[1].lowercase()) }
            else -> emptyList()
        }
    }

    override fun onCommand(sender: CommandSender, command: Command, label: String, args: Array<out String>): Boolean {
        when (args.getOrNull(0)?.lowercase()) {
            "menu" -> {
                if (sender !is Player) { sender.sendMessage(lang.get("ifz.in_game_only", sender)); return true }
                val menuId = args.getOrNull(1) ?: run { sender.sendMessage(lang.get("ifz.usage_menu", sender)); return true }
                plugin.menuProvider.openMenu(sender, menuId, lang.playerLang(sender))
            }
            "reload" -> {
                plugin.menuProvider.unload()
                plugin.menuProvider.load()
                sender.sendMessage(lang.get("ifz.reloaded", sender))
            }
            "sync" -> {
                when (args.getOrNull(1)?.lowercase()) {
                    "status" -> sender.sendMessage(lang.get("ifz.sync_status", sender))
                    "reject" -> {
                        plugin.menuProvider.unload()
                        plugin.menuProvider.load()
                        sender.sendMessage(lang.get("ifz.sync_reconnecting", sender))
                    }
                    else -> sender.sendMessage(lang.get("ifz.usage_sync", sender))
                }
            }
            "setup" -> setup(sender, args.getOrNull(1)?.uppercase())
            "editor" -> {
                when (args.getOrNull(1)?.lowercase()) {
                    "revoke" -> editorRevoke(sender)
                    null     -> editor(sender)
                    else     -> sender.sendMessage(lang.get("ifz.usage_editor", sender))
                }
            }
            "trusteditor" -> {
                val code = args.getOrNull(1) ?: run {
                    sender.sendMessage(lang.get("ifz.usage_trusteditor", sender))
                    return true
                }
                if (!sender.hasPermission("artisan.editor")) {
                    sender.sendMessage(lang.get("ifz.trust_no_permission", sender))
                    return true
                }
                plugin.menuProvider.sendTrustRequest(code.uppercase())
                sender.sendMessage(lang.get("ifz.trust_sent", sender))
            }
            else -> sender.sendMessage(lang.get("ifz.usage_main", sender))
        }
        return true
    }

        val cfg = PluginConfig.from(plugin.config)
        val senderLang = lang.playerLang(sender)
        if (cfg.editorConsoleOnly && sender !is ConsoleCommandSender) {
            sender.sendMessage(lang.get("ifz.editor_console_only", senderLang))
            return
        }
        if (!sender.hasPermission("artisan.editor")) {
            sender.sendMessage(lang.get("ifz.editor_no_permission", senderLang))
            return
        }
        if (cfg.wsUrl.isEmpty() || cfg.serverKey.isEmpty()) {
            sender.sendMessage(lang.get("ifz.server_not_linked", senderLang))
            return
        }
        val senderName = if (sender is Player) sender.name else "CONSOLE"
        plugin.menuProvider.requestEditorSession(senderName) { url, expiresAt ->
            plugin.logger.info("Editor → $url  (expires $expiresAt)")
            Bukkit.getScheduler().runTask(plugin, Runnable {
                val link = Component.text(lang.getRaw("ifz.editor_link_label", senderLang))
                    .color(NamedTextColor.GREEN)
                    .decorate(TextDecoration.UNDERLINED)
                    .clickEvent(ClickEvent.openUrl(url))
                sender.sendMessage(lang.get("ifz.prefix", senderLang).append(link))
                sender.sendMessage(lang.get("ifz.editor_expires", senderLang, "expires" to expiresAt))
            })
        }
    }

    private fun editorRevoke(sender: CommandSender) {

            sender.sendMessage(lang.get("ifz.editor_no_permission", sender))
            return
        }
        plugin.menuProvider.revokeEditorSessions()
        sender.sendMessage(lang.get("ifz.editor_revoked", sender))
    }

    private fun setup(sender: CommandSender, code: String?) {
        setupClient?.disconnect()
        setupClient = null
        val cfg = PluginConfig.from(plugin.config)
        val senderLang = lang.playerLang(sender)
        if (cfg.wsUrl.isEmpty()) {
            sender.sendMessage(lang.get("ifz.setup_ws_missing", senderLang))
            return
        }
        sender.sendMessage(lang.get("ifz.setup_connecting", senderLang))
        val configFile = plugin.dataFolder.resolve("config.yml")
        setupClient = SetupClient(
            url = cfg.wsUrl,
            code = code,
            onCode = { _, url ->
                plugin.logger.info("Setup link: $url")
                Bukkit.getScheduler().runTask(plugin, Runnable {
                    val link = Component.text(lang.getRaw("ifz.setup_link_label", senderLang))
                        .color(NamedTextColor.AQUA)
                        .decorate(TextDecoration.UNDERLINED)
                        .clickEvent(ClickEvent.openUrl(url))
                    sender.sendMessage(lang.get("ifz.prefix", senderLang).append(link))
                })
            },
            onLinked = { serverKey ->
                ConfigWriter.writeServerKey(configFile, serverKey)
                plugin.logger.info("Server linked (server_key written to config.yml)")
                Bukkit.getScheduler().runTask(plugin, Runnable {
                    sender.sendMessage(lang.get("ifz.setup_linked", senderLang))
                    plugin.reloadConfig()
                    plugin.reinit(PluginConfig.from(plugin.config))
                    sender.sendMessage(lang.get("ifz.setup_ready", senderLang))
                })
            },
        )
        setupClient!!.connect()
    }
}
```
